% Dumbbell (before->after) replacement for the old §6 line chart (2026-08-17
% chart redesign). Rationale: the old figure had four "lines" spanning only
% two x-positions (R1, R2) that nearly collide at R2 (Market A0 4.966 vs
% CUHK03 A3 4.977) -- that is a before/after comparison wearing line-chart
% grammar, not a trend. A dumbbell says exactly that: one dot per stage, one
% arrow, and the claim ("every arrow points right") is visually unmistakable.
%
% Two panels, one row per dataset x attacker, light dot = R1, dark dot = R2,
% same two navy shades used in the §5 heatmap ramp (navyramp2 / navyramp5 ==
% navy) so the whole poster's numeric charts read as one system. Row order is
% identical in both panels and matches the paper's tab:cuhk_replication order.
%
% Data source: paper tab:cuhk_replication. Cross-checked value-for-value
% against the paper table; see BUILD_NOTES.md. Axis maxima (0.80, 6.6) match
% the old figure's ymax so nothing is renormalized relative to the prior view.
\begin{tikzpicture}
\def\rowLabelW{3.0}\def\rowH{1.05}\def\topPad{1.6}\def\axisW{8.0}

% ---- legend (light=R1, dark=R2), once per panel ----
\newcommand{\dlegend}[1]{%
  \draw[fill=navyramp2, draw=navy, line width=0.8pt] ({#1+0.0},{-0.75}) circle (0.09);
  \node[anchor=west, font=\fsSmall\selectfont, color=navy] at ({#1+0.20},{-0.75}) {R1};
  \draw[fill=navyramp5, draw=navy, line width=0.8pt] ({#1+1.55},{-0.75}) circle (0.10);
  \node[anchor=west, font=\fsSmall\selectfont, color=navy] at ({#1+1.76},{-0.75}) {R2};
}

% ---- panel scaffold: #1 xOffset #2 title #3 xmax #4 metric-axis-label #5 tick-list ----
\newcommand{\dpanel}[5]{%
  \node[anchor=north, font=\fsSmall\bfseries\selectfont, color=navy] at (#1+0.5*(\rowLabelW+\axisW),0) {#2};
  \dlegend{#1+\rowLabelW}
  \draw[navy, line width=0.9pt] ({#1+\rowLabelW},{-\topPad-4*\rowH}) -- ++({\axisW},0);
  \foreach \tv in {#5} {
    \pgfmathsetmacro{\tx}{#1+\rowLabelW+(\tv)/(#3)*\axisW}
    \draw[gray!20, line width=0.5pt] (\tx,{-\topPad}) -- (\tx,{-\topPad-4*\rowH});
    \draw[navy, line width=0.7pt] (\tx,{-\topPad-4*\rowH}) -- ++(0,-0.07);
    \node[font=\fsSmall\selectfont, color=navy] at (\tx,{-\topPad-4*\rowH-0.32}) {\tv};
  }
  \node[font=\fsSmall\selectfont, color=navy] at ({#1+\rowLabelW+0.5*\axisW},{-\topPad-4*\rowH-1.02}) {#4};
}

% ---- one dumbbell row: #1 xOffset #2 scale(cm/unit) #3 rowIndex #4 label-line1 #5 label-line2 #6 v1(R1) #7 v2(R2) #8 v1text #9 v2text ----
\newcommand{\drow}[9]{%
  \pgfmathsetmacro{\ry}{-\topPad-0.5*\rowH-(#3)*\rowH}
  \pgfmathsetmacro{\rxone}{#1+\rowLabelW+(#6)*(#2)}
  \pgfmathsetmacro{\rxtwo}{#1+\rowLabelW+(#7)*(#2)}
  \node[anchor=east, font=\fsSmall\selectfont, color=navy] at ({#1+\rowLabelW-0.18},{\ry}) {#4~#5};
  \draw[navy, line width=1.7pt, -{Latex[length=2.4mm,width=1.8mm]}] (\rxone,\ry) -- (\rxtwo,\ry);
  \draw[fill=navyramp2, draw=navy, line width=0.9pt] (\rxone,\ry) circle (0.13);
  \draw[fill=navyramp5, draw=navy, line width=0.9pt] (\rxtwo,\ry) circle (0.15);
  % Labels flank the pair (R1 label up-left of its dot, R2 label up-right of
  % its dot) rather than both sitting directly above -- this keeps them from
  % colliding when the two dots are close together (small R1->R2 gap).
  \node[anchor=south east, font=\fsSmall\selectfont, color=muted] at ({\rxone-0.10},{\ry+0.20}) {#8};
  \node[anchor=south west, font=\fsSmall\selectfont, color=navy] at ({\rxtwo+0.10},{\ry+0.20}) {\textbf{#9}};
}
